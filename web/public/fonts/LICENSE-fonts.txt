import sys
from fontTools.ttLib import TTFont
from fontTools.varLib import instancer
from fontTools import subset

D = "/tmp/atelier-fonts/dl"
JOBS = [
  (f"{D}/sans/VF/SourceSans3VF-Upright.ttf", "AtelierSans-VF.woff2", "Atelier Sans", "Regular", {"wght": (400, 600)}),
  (f"{D}/sans/VF/SourceSans3VF-Italic.ttf", "AtelierSans-Italic-VF.woff2", "Atelier Sans", "Italic", {"wght": (400, 600)}),
  (f"{D}/serif/source-serif-4.005_Desktop/VAR/SourceSerif4Variable-Roman.ttf", "AtelierSerif-VF.woff2", "Atelier Serif", "Regular", {"wght": (400, 600), "opsz": (16, 36)}),
  (f"{D}/code/VF/SourceCodeVF-Upright.ttf", "AtelierCode-VF.woff2", "Atelier Code", "Regular", {"wght": (400, 500)}),
]
UNI = "U+0000-017F,U+0192,U+02C6-02DC,U+2000-206F,U+20A0-20CF,U+2100-214F,U+2190-21FF,U+2200-22FF,U+2300-23FF,U+2460-24FF,U+2500-25FF,U+2700-27BF,U+FFFD"
FEAT = "kern,liga,calt,tnum,lnum,smcp,c2sc,case"

def rename(font, family, style):
    ps = family.replace(" ", "") + ("-" + style if style != "Regular" else "")
    name = font["name"]
    for rec in list(name.names):
        if rec.nameID in (1, 4, 16, 21):
            rec.string = family if rec.nameID in (1, 16, 21) else f"{family} {style}".replace(" Regular", "")
        elif rec.nameID in (2, 17, 22):
            rec.string = style
        elif rec.nameID == 3:
            rec.string = f"{ps};atelier-subset"
        elif rec.nameID == 6:
            rec.string = ps
        elif rec.nameID == 25:
            rec.string = family.replace(" ", "")
    # instance/axis names (nameID >= 256) are style names like "Bold", fine to keep

for src, out, family, style, limits in JOBS:
    f = TTFont(src)
    present = {a.axisTag for a in f["fvar"].axes}
    f = instancer.instantiateVariableFont(f, {k: v for k, v in limits.items() if k in present})
    rename(f, family, style)
    tmp = f"/tmp/atelier-fonts/r2/{out}.ttf"
    f.save(tmp)
    f = TTFont(tmp)
    opts = subset.Options()
    opts.flavor = "woff2"
    opts.layout_features = FEAT.split(",")
    opts.name_IDs = ["*"]
    opts.name_languages = ["*"]
    opts.notdef_outline = True
    s = subset.Subsetter(opts)
    s.populate(unicodes=subset.parse_unicodes(UNI))
    s.subset(f)
    f.flavor = "woff2"
    f.save(f"/tmp/atelier-fonts/r2/{out}")
    axes = [(a.axisTag, a.minValue, a.maxValue) for a in f["fvar"].axes]
    print(out, axes, f["name"].getDebugName(1), f["name"].getDebugName(6))
